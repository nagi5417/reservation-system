## 認証（Authentication）

### 認証方式

#### 1. Google OAuth 2.0

**フロー**:
```
User → Google認証画面 → 認証成功 → Callback → Spring Security
→ OAuth2UserService → User登録/ログイン → Session生成
```

**設定**（application.yml）:
```yaml
spring:
  security:
    oauth2:
      client:
        registration:
          google:
            client-id: ${GOOGLE_CLIENT_ID}
            client-secret: ${GOOGLE_CLIENT_SECRET}
            scope:
              - openid
              - profile
              - email
              - https://www.googleapis.com/auth/calendar.events
            redirect-uri: "{baseUrl}/login/oauth2/code/{registrationId}"
        provider:
          google:
            authorization-uri: https://accounts.google.com/o/oauth2/v2/auth
            token-uri: https://oauth2.googleapis.com/token
            user-info-uri: https://www.googleapis.com/oauth2/v3/userinfo
            user-name-attribute: sub
```

**OAuth2UserService実装**:
```java
package com.example.reservation.security;

import com.example.reservation.entity.User;
import com.example.reservation.entity.enums.UserRole;
import com.example.reservation.repository.UserRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.security.oauth2.client.userinfo.DefaultOAuth2UserService;
import org.springframework.security.oauth2.client.userinfo.OAuth2UserRequest;
import org.springframework.security.oauth2.core.OAuth2AuthenticationException;
import org.springframework.security.oauth2.core.user.OAuth2User;
import org.springframework.stereotype.Service;

@Service
@RequiredArgsConstructor
public class CustomOAuth2UserService extends DefaultOAuth2UserService {

    private final UserRepository userRepository;

    @Override
    public OAuth2User loadUser(OAuth2UserRequest userRequest)
        throws OAuth2AuthenticationException {

        OAuth2User oAuth2User = super.loadUser(userRequest);

        // Google OAuth属性取得
        String googleSub = oAuth2User.getAttribute("sub");
        String email = oAuth2User.getAttribute("email");
        String name = oAuth2User.getAttribute("name");

        // トークン取得
        String accessToken = userRequest.getAccessToken().getTokenValue();
        String refreshToken = userRequest.getAdditionalParameters()
            .get("refresh_token") != null
            ? userRequest.getAdditionalParameters().get("refresh_token").toString()
            : null;

        // ユーザー検索または作成
        User user = userRepository.findByGoogleSub(googleSub)
            .orElseGet(() -> createGoogleUser(googleSub, email, name));

        // トークン更新
        user.setGoogleAccessToken(accessToken);
        user.setGoogleRefreshToken(refreshToken);
        userRepository.save(user);

        return new CustomUserDetails(user, oAuth2User.getAttributes());
    }

    private User createGoogleUser(String googleSub, String email, String name) {
        User user = User.builder()
            .googleSub(googleSub)
            .email(email)
            .name(name)
            .role(UserRole.USER)
            .isEmailVerified(true) // Google認証済み
            .build();
        return userRepository.save(user);
    }
}
```

#### 2. メール/パスワード認証

**UserDetailsService実装**:
```java
package com.example.reservation.security;

import com.example.reservation.entity.User;
import com.example.reservation.repository.UserRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.security.core.userdetails.UserDetails;
import org.springframework.security.core.userdetails.UserDetailsService;
import org.springframework.security.core.userdetails.UsernameNotFoundException;
import org.springframework.stereotype.Service;

@Service
@RequiredArgsConstructor
public class CustomUserDetailsService implements UserDetailsService {

    private final UserRepository userRepository;

    @Override
    public UserDetails loadUserByUsername(String email)
        throws UsernameNotFoundException {

        User user = userRepository.findByEmail(email)
            .orElseThrow(() -> new UsernameNotFoundException(
                "ユーザーが見つかりません: " + email
            ));

        // メールアドレス確認チェック
        if (!user.getIsEmailVerified()) {
            throw new UsernameNotFoundException("メールアドレスが確認されていません");
        }

        return new CustomUserDetails(user);
    }
}
```

**CustomUserDetails**:
```java
package com.example.reservation.security;

import com.example.reservation.entity.User;
import com.example.reservation.entity.enums.UserRole;
import lombok.Getter;
import org.springframework.security.core.GrantedAuthority;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.core.userdetails.UserDetails;
import org.springframework.security.oauth2.core.user.OAuth2User;

import java.util.Collection;
import java.util.Collections;
import java.util.Map;

@Getter
public class CustomUserDetails implements UserDetails, OAuth2User {

    private final User user;
    private Map<String, Object> attributes;

    // メール/パスワード認証用
    public CustomUserDetails(User user) {
        this.user = user;
    }

    // Google OAuth用
    public CustomUserDetails(User user, Map<String, Object> attributes) {
        this.user = user;
        this.attributes = attributes;
    }

    public Long getUserId() {
        return user.getId();
    }

    @Override
    public String getName() {
        return user.getName();
    }

    @Override
    public Collection<? extends GrantedAuthority> getAuthorities() {
        return Collections.singletonList(
            new SimpleGrantedAuthority("ROLE_" + user.getRole().name())
        );
    }

    @Override
    public String getPassword() {
        return user.getPassword();
    }

    @Override
    public String getUsername() {
        return user.getEmail();
    }

    @Override
    public boolean isAccountNonExpired() {
        return true;
    }

    @Override
    public boolean isAccountNonLocked() {
        return true;
    }

    @Override
    public boolean isCredentialsNonExpired() {
        return true;
    }

    @Override
    public boolean isEnabled() {
        return user.getIsEmailVerified();
    }

    @Override
    public Map<String, Object> getAttributes() {
        return attributes;
    }
}
```

## 認可（Authorization）

### ロールベースアクセス制御（RBAC）

#### ロール定義

| ロール | 権限 |
|--------|------|
| USER | 予約作成、自分の予約閲覧・キャンセル |
| STAFF | 予約枠作成・編集、全予約閲覧、メニュー管理 |
| ADMIN | スタッフ管理（MVP外） |

---

#### メソッドレベル認可

**@PreAuthorize使用例**:

```java
// STAFF専用
@PreAuthorize("hasRole('STAFF')")
@PostMapping("/api/staff/slots")
public ResponseEntity<SlotResponse> createSlot(...) {
    // ...
}

// USER専用
@PreAuthorize("hasRole('USER')")
@PostMapping("/api/reservations")
public ResponseEntity<ReservationResponse> createReservation(...) {
    // ...
}

// 認証必須（ロール不問）
@PreAuthorize("isAuthenticated()")
@GetMapping("/api/users/me")
public ResponseEntity<UserResponse> getMe(...) {
    // ...
}

// 公開（認証不要）
@GetMapping("/api/slots")
public ResponseEntity<List<SlotResponse>> getPublicSlots(...) {
    // ...
}
```

#### URL パターンベース認可

```java
@EnableWebSecurity

                .anyRequest().authenticated()
            );

        return http.build();
    }
}
```

## パスワードセキュリティ

### BCrypt暗号化

**設定**:
```java
@Bean
public PasswordEncoder passwordEncoder() {
    return new BCryptPasswordEncoder(10); // コスト係数10
}
```

**使用例**:
```java
@Service
public class AuthService {

    private final PasswordEncoder passwordEncoder;

    // 新規登録時
    public void signup(SignupRequest request) {
        String encodedPassword = passwordEncoder.encode(request.getPassword());

        User user = User.builder()
            .email(request.getEmail())
            .password(encodedPassword) // 暗号化済みパスワード
            .name(request.getName())
            .build();

        userRepository.save(user);
    }

    // ログイン時（Spring Securityが自動検証）
    // passwordEncoder.matches(rawPassword, encodedPassword) が内部で呼ばれる
}
```

**セキュリティ特性**:
- ソルト自動生成（同じパスワードでも毎回異なるハッシュ）
- コスト係数10 = 2^10回のハッシュ計算（総当たり攻撃への耐性）
- 一方向暗号化（復号不可能）

**保存例**:
```
入力: password123
出力: $2a$10$N9qo8uLOickgx2ZMRZoMyeIjZAgcfl7p92ldGxad68LJZdL17lhWy
```

## CSRF対策

### Spring Security CSRF保護

**設定**:
```java
@Bean
public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
    http
        .csrf(csrf -> csrf
            .csrfTokenRepository(CookieCsrfTokenRepository.withHttpOnlyFalse())
        );

    return http.build();
}
```

**動作**:
1. 初回アクセス時、サーバーがCSRFトークンをCookieに設定
2. フロントエンドがトークンを取得し、リクエストヘッダーに含める
3. サーバーがトークンを検証

**フロントエンド実装（React + Axios）**:
```javascript
// Axiosインターセプター設定
axios.interceptors.request.use(config => {
  const token = getCookie('XSRF-TOKEN');
  if (token) {
    config.headers['X-XSRF-TOKEN'] = token;
  }
  return config;
});

function getCookie(name) {
  const value = `; ${document.cookie}`;
  const parts = value.split(`; ${name}=`);
  if (parts.length === 2) return parts.pop().split(';').shift();
}
```

## CORS設定

### CorsConfig

```java
package com.example.reservation.config;

import org.springframework.context.annotation.Configuration;
import org.springframework.web.servlet.config.annotation.CorsRegistry;
import org.springframework.web.servlet.config.annotation.WebMvcConfigurer;

@Configuration
public class CorsConfig implements WebMvcConfigurer {

    @Override
    public void addCorsMappings(CorsRegistry registry) {
        registry.addMapping("/api/**")
            .allowedOrigins(getAllowedOrigins())
            .allowedMethods("GET", "POST", "PUT", "PATCH", "DELETE", "OPTIONS")
            .allowedHeaders("*")
            .allowCredentials(true) // Cookie/セッション送信を許可
            .maxAge(3600); // プリフライトリクエストのキャッシュ時間
    }

    private String[] getAllowedOrigins() {
        // 環境変数で切り替え
        String env = System.getenv("ENV");
        if ("production".equals(env)) {
            return new String[]{"https://yourdomain.com"};
        } else {
            return new String[]{"http://localhost:3000"}; // React開発サーバー
        }
    }
}
```

## SQLインジェクション対策

### JPA/Hibernateのパラメータバインディング

**安全な実装（推奨）**:
```java
@Repository
public interface UserRepository extends JpaRepository<User, Long> {

    // Spring Data JPAのメソッド命名規則（安全）
    Optional<User> findByEmail(String email);

    // JPQLパラメータバインディング（安全）
    @Query("SELECT u FROM User u WHERE u.email = :email")
    Optional<User> findByEmailCustom(@Param("email") String email);
}
```

**危険な実装（絶対NG）**:
```java
// ❌ 文字列連結（SQLインジェクションの危険）
@Query(value = "SELECT * FROM users WHERE email = '" + email + "'", nativeQuery = true)
Optional<User> findByEmailUnsafe(String email);
```

**プリペアドステートメント使用（安全）**:
```java
// ネイティブクエリでもパラメータバインディングを使用
@Query(value = "SELECT * FROM users WHERE email = ?1", nativeQuery = true)
Optional<User> findByEmailNative(String email);
```

### バックエンド対策

- Spring BootのJacksonが自動的にエスケープ

```java
@GetMapping("/api/users/{id}")

    return userService.getUser(id); // JSON変換時に自動エスケープ
}
```

## Spring Security設定詳細

### SecurityConfig.java（完全版）

```java
package com.example.reservation.config;

import com.example.reservation.security.CustomOAuth2UserService;
import com.example.reservation.security.CustomUserDetailsService;
import com.example.reservation.security.OAuth2SuccessHandler;
import lombok.RequiredArgsConstructor;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.config.annotation.method.configuration.EnableMethodSecurity;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.security.web.csrf.CookieCsrfTokenRepository;

@Configuration
@EnableWebSecurity
@EnableMethodSecurity // @PreAuthorizeを有効化
@RequiredArgsConstructor
public class SecurityConfig {

    private final CustomUserDetailsService userDetailsService;
    private final CustomOAuth2UserService oAuth2UserService;
    private final OAuth2SuccessHandler oAuth2SuccessHandler;

    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
        http
            // 認可設定
            .authorizeHttpRequests(auth -> auth
                // 公開エンドポイント
                .requestMatchers("/api/auth/**").permitAll()
                .requestMatchers("/api/slots").permitAll()
                .requestMatchers("/api/slots/{slotId}").permitAll()
                .requestMatchers("/api/menus").permitAll()
                .requestMatchers("/swagger-ui/**", "/v3/api-docs/**").permitAll()
                .requestMatchers("/error").permitAll()

                // STAFF専用
                .requestMatchers("/api/staff/**").hasRole("STAFF")

                // その他は認証必須
                .anyRequest().authenticated()
            )

            // フォームログイン（メール/パスワード）
            .formLogin(form -> form
                .loginProcessingUrl("/api/auth/login")
                .successHandler((request, response, authentication) -> {
                    response.setStatus(200);
                })
                .failureHandler((request, response, exception) -> {
                    response.setStatus(401);
                })
            )

            // ログアウト
            .logout(logout -> logout
                .logoutUrl("/api/auth/logout")
                .logoutSuccessHandler((request, response, authentication) -> {
                    response.setStatus(204);
                })
                .deleteCookies("JSESSIONID")
            )

            // OAuth2ログイン（Google）
            .oauth2Login(oauth2 -> oauth2
                .loginPage("/api/auth/google")
                .userInfoEndpoint(userInfo -> userInfo
                    .userService(oAuth2UserService)
                )
                .successHandler(oAuth2SuccessHandler)
            )

            // CSRF対策
            .csrf(csrf -> csrf
                .csrfTokenRepository(CookieCsrfTokenRepository.withHttpOnlyFalse())
            )

            // セキュリティヘッダー
            .headers(headers -> headers
                .frameOptions(frame -> frame.deny())
                .xssProtection(xss -> xss.block(true))
            )

            // UserDetailsService設定
            .userDetailsService(userDetailsService);

        return http.build();
    }

    @Bean
    public PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder(10);
    }
}
```
